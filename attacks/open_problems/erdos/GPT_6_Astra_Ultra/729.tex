\section{Erd\H{o}s Problem \#729: only bounded primes in a factorial denominator}
\noindent A complete carry-and-valuation argument, 24 September 2026.

\subsection*{Statement and parameters}
Given any fixed $C>0$, we construct infinitely many $a,b,n$ with
$a+b>n+C\log n$ for which the reduced denominator of $n!/(a!b!)$ has no prime divisor exceeding a constant depending only on $C$. Our choices also satisfy $a/n,b/n\to1/2$.
Put $D=C+1$, and for a large integer $M$ write
\[
 L=\log M,\qquad K=\lceil DL\rceil,\qquad M\leq m<2M.
\]
We will take $n=2m$, $a=m+K$, $b=m$.

\subsection*{A valuation bound for consecutive factors}
Let $\kappa_p(m)=v_p\binom{2m}{m}$ and $J_p=\max_{1\leq i\leq K}v_p(m+i)$. Counting multiples of $p^j$, after separating a term attaining $J_p$, yields
\[
 W_p:=v_p((m+1)\cdots(m+K))
 \leq v_p(K!)+J_p
 \leq\frac K{p-1}+J_p. \tag{1}
\]
For clarity, at each $j\leq J_p$ the other multiples number at most
$\lceil K/p^j\rceil-1\leq\lfloor K/p^j\rfloor$; for larger $j$ there are none. Summing proves the first inequality. The geometric series $\sum_{j\geq1}K/p^j=K/(p-1)$ proves the second.

\subsection*{Simultaneous carry and spike estimates}
We prove that all but $o_C(M)$ choices of $m$ satisfy, for every prime $p\leq2K$,
\[
 \kappa_p(m)\geq\frac L{18\log p},
 \qquad
 J_p\leq\left\lceil\frac{4\log L}{\log p}\right\rceil. \tag{2}
\]
Legendre's formula gives
\[
 \kappa_p(m)=\sum_{j\geq1}
 \left(\left\lfloor2m/p^j\right\rfloor
       -2\left\lfloor m/p^j\right\rfloor\right).
\]
Each digit at least $\lceil p/2\rceil$ forces the corresponding summand to be one.
Let $d=\lfloor L/(2\log p)\rfloor$. Among uniform residues modulo $p^d$, the number of such digits is binomial with success probability at least $1/3$. Exponential Markov inequality, or its standard binomial consequence, bounds the probability of fewer than $d/6$ successes by $e^{-d/24}$.
The modulus is at most $\sqrt M$, so at most $2M e^{-d/24}$ integers in $[M,2M)$ fail. For all $p\leq2K$, $d\geq L/(3\log p)$ once $M$ is large. The total failure count is at most
\[
 4KM\exp\left(-\frac L{72\log(2K)}\right)=o_C(M).
\]
For the second inequality, if $T_p=\lceil4\log L/\log p\rceil$, the event $v_p(m+i)>T_p$ has at most $M/L^4+1$ solutions in that interval. Summing over $p\leq2K$ and $i\leq K$ gives at most
$2K^2(M/L^4+1)=o_C(M)$. This proves (2), without independence assumptions between primes.

\subsection*{A fixed cutoff and the medium primes}
Choose a constant $P=P(C)$ so large that
\[
 \frac{(D+1)\log p}{p-1}\leq\frac1{40}\qquad(p\geq P).
\]
Such a choice exists since $\log p/(p-1)\to0$.
For large $M$, we have $K\leq(D+1)L$, and uniformly for $p\leq2K$,
\[
 4\log L+\log p\leq4\log L+\log(2K)\leq L/40.
\]
For a good $m$ and $P\leq p\leq2K$, (1)--(2) therefore imply
\[
 W_p\leq\frac{L}{40\log p}+\frac{L}{40\log p}
       =\frac{L}{20\log p}
       <\kappa_p(m). \tag{3}
\]

\subsection*{All larger primes}
If $p>2K$, at most one of $m+1,\ldots,m+K$ is divisible by $p$. Hence $W_p=J_p$. If its maximizing term is $m+i$ and $J_p=J>0$, then for every $j\leq J$,
$m\bmod p^j=p^j-i>p^j/2$. Each such level contributes one in Legendre's formula, so $\kappa_p(m)\geq J=W_p$. If $J=0$ the same conclusion is immediate.

The rational number in the question is
\[
 \frac{n!}{a!b!}
 =\frac{\binom{2m}{m}}{(m+1)\cdots(m+K)}.
\]
Equations (3) and the large-prime argument show that its valuation is nonnegative for every prime $p\geq P$. Its reduced denominator therefore has only primes below $P$, a bound independent of $m$.

\subsection*{Excess, infinitude and attribution}
For large $M$,
\[
 a+b-n=K\geq(C+1)\log M>C\log(2m)=C\log n.
\]
There is a good $m$ in every sufficiently large dyadic interval, giving infinitely many distinct triples. Moreover $K/m\to0$, so both $a,b$ are asymptotically half of $n$.
