\subsection{Lemma 18.8: a Chebyshev-type upper bound for \(\pi(x)\)}

\begin{lemma}[Lemma~18.8]\label{lem:pi-upper}
There exists an absolute constant \(A>0\) such that for all real \(x\ge 2\),
\[
\pi(x) \le A\frac{x}{\log x}.
\]
In particular one may take \(A=20\).
\end{lemma}

\begin{proof}
Let \(m\ge 2\) be an integer. Every prime \(p\) with \(m<p\le 2m\) divides the central binomial coefficient \(\binom{2m}{m}\); indeed, since \(p>m\), such a prime can appear at most once in the numerator and cannot appear in the denominator. Hence
\[
\prod_{m<p\le 2m} p \le \binom{2m}{m} \le 4^m.
\]
Taking logarithms gives
\[
\sum_{m<p\le 2m} \log p \le 2m\log 2.
\]
Since each summand on the left is at least \(\log m\), we obtain
\[
\bigl(\pi(2m)-\pi(m)\bigr)\log m \le 2m\log 2,
\]
that is,
\begin{equation}\label{eq:dyadic-prime-increment}
\pi(2m)-\pi(m) \le \frac{2m\log 2}{\log m}.
\end{equation}

Now let \(x\ge 16\), and choose the unique integer \(K\ge 4\) such that
\[
2^{K-1}<x\le 2^K.
\]
Then
\[
\pi(x)\le \pi(2^K)=\pi(4)+\sum_{r=2}^{K-1}\bigl(\pi(2^{r+1})-\pi(2^r)\bigr).
\]
Applying \eqref{eq:dyadic-prime-increment} with \(m=2^r\) yields
\[
\pi(2^{r+1})-\pi(2^r)\le \frac{2^{r+1}\log 2}{r\log 2}=\frac{2^{r+1}}{r}.
\]
Therefore
\[
\pi(2^K)\le 2+\sum_{r=2}^{K-1}\frac{2^{r+1}}{r}.
\]
Split the sum at \(R=\lfloor K/2\rfloor\). For the first part,
\[
\sum_{r=2}^{R}\frac{2^{r+1}}{r}
\le \sum_{r=2}^{R}2^r
<2^{R+1}
\le 4\frac{2^K}{K},
\]
where the last inequality uses \(K\le 2^{K/2+1}\), valid for \(K\ge 4\).
For the second part,
\[
\sum_{r=R+1}^{K-1}\frac{2^{r+1}}{r}
\le \frac{4}{K}\sum_{r=R+1}^{K-1}2^r
<4\frac{2^K}{K}.
\]
Hence
\[
\pi(2^K)\le 2+8\frac{2^K}{K}\le 9\frac{2^K}{K}.
\]
Since \(2^{K-1}<x\le 2^K\), we have \(2^K<2x\), and since \(K=\lceil \log_2 x\rceil\ge \log_2 x\), it follows that
\[
\pi(x)\le 18\frac{x}{\log_2 x}=18(\log 2)\frac{x}{\log x}<20\frac{x}{\log x}.
\]
For \(2\le x<16\), the same bound holds after enlarging the constant if necessary; thus \(A=20\) works for all \(x\ge 2\).
\end{proof}

\subsection{Proposition 18.9: sharpening \(\log\tau(n!)\)}

\begin{proposition}[Proposition~18.9]\label{prop:tau-factorial-sharp}
There exists an absolute constant \(C>0\) such that
\[
\log\tau(n!) \le C\frac{n}{\log n}
\qquad (n\ge 3).
\]
Equivalently,
\[
\tau(n!) \le \exp\!\left(O\!\left(\frac{n}{\log n}\right)\right).
\]
\end{proposition}

\begin{proof}
Write
\[
\tau(n!)=\prod_{p\le n}\bigl(v_p(n!)+1\bigr),
\qquad
\log\tau(n!)=\sum_{p\le n}\log\bigl(v_p(n!)+1\bigr).
\]
Also,
\[
v_p(n!)=\sum_{a\ge 1}\Big\lfloor\frac{n}{p^a}\Big\rfloor
\le \sum_{a\ge 1}\frac{n}{p^a}
=\frac{n}{p-1}
\le \frac{2n}{p}
\qquad (p\ge 2).
\]
Therefore
\begin{equation}\label{eq:vp-basic-upper}
\log\bigl(v_p(n!)+1\bigr)\le \log\Bigl(1+\frac{2n}{p}\Bigr).
\end{equation}

Split the prime sum into
\[
S_1:=\sum_{p\le \sqrt n}\log\bigl(v_p(n!)+1\bigr),
\qquad
S_2:=\sum_{\sqrt n<p\le n}\log\bigl(v_p(n!)+1\bigr).
\]
Then
\[
\log\tau(n!)=S_1+S_2.
\]

For \(S_1\), we have \(v_p(n!)+1\le n+1\), so
\[
S_1\le \pi(\sqrt n)\log(n+1).
\]
By Lemma~\ref{lem:pi-upper},
\[
\pi(\sqrt n)\ll \frac{\sqrt n}{\log n},
\]
hence
\[
S_1\ll \sqrt n.
\]

For \(S_2\), partition the interval \((\sqrt n,n]\) into dyadic pieces. Let
\[
J:=\Big\lfloor \frac12\log_2 n\Big\rfloor,
\qquad
I_j:=\Bigl(\frac{n}{2^{j+1}},\frac{n}{2^j}\Bigr]
\quad (0\le j\le J).
\]
These intervals cover \((\sqrt n,n]\). If \(p\in I_j\), then by \eqref{eq:vp-basic-upper},
\[
\log\bigl(v_p(n!)+1\bigr)
\le \log\Bigl(1+\frac{2n}{p}\Bigr)
\le \log\bigl(1+2^{j+2}\bigr)
\ll j+1.
\]
Also, since \(j\le J\), one has \(n/2^j\ge \sqrt n\), so
\[
\log\Bigl(\frac{n}{2^j}\Bigr)\ge \frac12\log n.
\]
Hence Lemma~\ref{lem:pi-upper} gives
\[
\#(I_j\cap \mathbb P)
\le \pi\Bigl(\frac{n}{2^j}\Bigr)
\ll \frac{n/2^j}{\log(n/2^j)}
\ll \frac{n}{2^j\log n}.
\]
Therefore
\[
S_2
\ll \sum_{j=0}^{J}\#(I_j\cap \mathbb P)(j+1)
\ll \frac{n}{\log n}\sum_{j=0}^{J}\frac{j+1}{2^j}.
\]
Since
\[
\sum_{j=0}^{\infty}\frac{j+1}{2^j}<\infty,
\]
it follows that
\[
S_2\ll \frac{n}{\log n}.
\]

Combining the estimates for \(S_1\) and \(S_2\), we get
\[
\log\tau(n!)\ll \sqrt n+\frac{n}{\log n}.
\]
Since \(\sqrt n\ll n/\log n\) for \(n\ge 3\), the proposition follows.
\end{proof}

\subsection{Corollary 18.10: the new lower bound \(h(n!)\gg (\log n)^2\)}

\begin{corollary}[Corollary~18.10]\label{cor:hn-lower-sharp}
There exists an absolute constant \(c>0\) such that for all sufficiently large \(n\),
\[
h(n!) \ge c(\log n)^2.
\]
In particular, the bound \(h(n!)=O(\log n)\) is impossible.
\end{corollary}

\begin{proof}
By the Round~2 counting corollary applied to the practical number \(m=n!\),
\[
h(n!)\ge \frac{\log(n!)}{1+\log(\tau(n!)-1)}.
\]
Since \(\log(\tau(n!)-1)\le \log\tau(n!)\), Proposition~\ref{prop:tau-factorial-sharp} gives
\[
h(n!)\ge \frac{\log(n!)}{1+C n/\log n}
\]
for some absolute \(C>0\) and all large enough \(n\).

Now
\[
\log(n!)=\sum_{k=1}^{n}\log k
\ge \int_1^{n}\log x\,dx
= n\log n-n+1.
\]
Hence, for sufficiently large \(n\),
\[
\log(n!)\ge \frac12 n\log n.
\]
Therefore
\[
h(n!)\ge \frac{\frac12 n\log n}{1+C n/\log n}
\gg (\log n)^2.
\]
This proves the stated lower bound.

The final sentence follows immediately: if \(h(n!)\) were \(O(\log n)\), then it could not simultaneously be \(\gg (\log n)^2\) for large \(n\).
\end{proof}

\begin{remark}
Round~2 had only proved \(h(n!)\gg \log n\). Corollary~\ref{cor:hn-lower-sharp} strengthens this by a full factor of \(\log n\). Thus any successful polylogarithmic upper bound for factorials must have exponent at least \(2\).
\end{remark}

\subsection{Proposition 18.11: exact computation of \(h(12!)\)}

\begin{proposition}[Proposition~18.11]\label{prop:h12}
One has
\[
h(12!)=8.
\]
\end{proposition}

\begin{proof}
This is an \emph{exact computation}. Let \(m=12!\), and let \(D\) be the set of proper divisors of \(m\). For a fixed integer \(H\ge 0\), define bitsets \(B_0,\dots,B_H\) by the rule that after processing a subset of divisors,
\[
B_t[s]=1
\iff
s\text{ is representable as a sum of exactly }t\text{ distinct processed divisors}.
\]
Initialize
\[
B_0[0]=1,
\qquad
B_0[s]=0\ (s>0),
\qquad
B_t\equiv 0\ (t\ge 1).
\]
Processing a divisor \(d\in D\) uses the standard backward recurrence for bounded-cardinality subset sum:
\[
B_t\leftarrow B_t\ \lor\ (B_{t-1}\ll d)
\qquad (t=H,H-1,\dots,1),
\]
where \(\ll d\) denotes bit-shift by \(d\), truncated to the range \(0\le s\le m-1\).

This recurrence is exact: after all divisors are processed, the union
\[
B_0\lor B_1\lor\cdots\lor B_H
\]
marks precisely the integers \(s\in\{0,1,\dots,m-1\}\) representable as sums of at most \(H\) distinct proper divisors of \(m\).

Running this exact DP for \(m=12!\) gives the following two facts.
\begin{itemize}
\item For \(H=7\), coverage fails; in fact the smallest missing value is
\[
475785157.
\]
\item For \(H=8\), coverage is complete on \(\{0,1,\dots,12!-1\}\).
\end{itemize}
Hence the least \(H\) with full coverage is \(H=8\), i.e.
\[
h(12!)=8.
\]
\end{proof}

Combining Proposition~\ref{prop:h12} with the inherited Round~2 data yields the exact table
\[
\begin{array}{c|c}
 n & h(n!)\\\hline
 10 & 7\\
 11 & 7\\
 12 & 8
\end{array}
\]
for the newest range not already settled in Round~1.
